### Complete Test Example
```javascript
describe('Visitor Management', () => {
  it('should handle visitor lifecycle with RBAC', async () => {
    // Create test data with Kenyan info
    const visitor = {
      name: generateKenyanName('Kikuyu'),
      phone: generateKenyanPhone(),
      address: generateKenyanAddress('Nairobi')
    };
    
    // Test all roles
    const testCases = generateRBACTestCases('visitors', 'create');
    
    for (const tc of testCases) {
      // Measure performance
      const { duration, result } = await measureResponseTime(async () => {
        return await request(app)
          .post('/api/visitors')
          .set('Authorization', `Bearer ${tc.token}`)
          .send(visitor);
      });
      
      // Validate response
      if (tc.shouldPass) {
        expect(result.status).toBe(201);
        expect(duration).toBeLessThan(200);
        assertSchema(result.body, { id: 'number', name: 'string' });
      } else {
        expect(result.status).toBe(403);
        assertErrorType(result.body, 'AuthorizationError');
      }
    }
  });
});
```
